    // Google Sign-In (Credential Manager)
    // ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??ƒ??
    private fun launchGoogleSignIn() {
        lifecycleScope.launch {
            try {
                setLoading(true)
                tvHint.text = "á?á?á?â? á?â?á?á?áæá?â? á?â? Google..."

                val credentialManager = CredentialManager.create(this@LoginActivity)

                val googleIdOption = GetSignInWithGoogleOption
                    .Builder(Constants.GOOGLE_WEB_CLIENT_ID)
                    .build()

                val request = GetCredentialRequest.Builder()
                    .addCredentialOption(googleIdOption)
                    .build()

                val result = credentialManager.getCredential(
                    context = this@LoginActivity,
                    request = request
                )

                val credential = result.credential

                if (credential is CustomCredential &&
                    credential.type == GoogleIdTokenCredential.TYPE_GOOGLE_ID_TOKEN_CREDENTIAL
                ) {
                    try {
                        val googleIdTokenCredential =
                            GoogleIdTokenCredential.createFrom(credential.data)
                        val idToken = googleIdTokenCredential.idToken
                        Log.d(TAG, "ƒ?? Got Google ID Token")
                        tvHint.text = ""
                        viewModel.loginWithGoogle(idToken, role)
                    } catch (e: GoogleIdTokenParsingException) {
                        setLoading(false)
                        tvHint.text = "á?áúáœ â?â? á?â?á?â?á?á? Google"
                    }
                } else {
                    setLoading(false)
                    tvHint.text = "â?â?á? á?â?á?â?á?á? á?â?á? â?á?á?â?â?"
                }

            } catch (e: GetCredentialCancellationException) {
                setLoading(false)
                tvHint.text = ""
            } catch (e: NoCredentialException) {
                setLoading(false)
                Log.e(TAG, "NoCredentialException: ${e.message}")
                tvHint.text = "â?á? â?â?á?á? á?á?á?á? Google â?á?á?á?"
            } catch (e: GetCredentialException) {
                setLoading(false)
                Log.e(TAG, "Google Credential Manager failed for role=$role", e)
                tvHint.text = "á?á?áøá? á?â?á?á?áæá?â? á?â? Google"
            } catch (e: Exception) {
                setLoading(false)
                Log.e(TAG, "Exception: ${e.message}")
                tvHint.text = "á?á?á® á?áúáœ áœá®â?á?á? á?â?á?á?â?â?"
            }
        }
    }
